--pouze na úrovni tabulky
CREATE TABLE clients4
(
client_number NUMBER(6),
first_name VARCHAR2(20),
last_name VARCHAR2(20),
phone VARCHAR2(20) CONSTRAINT phone_email_uk2 UNIQUE (email, phone), -- chyba 1 - nelze provádět na úrovníi sloupce omezení pro více sloupců najednou
email VARCHAR2(10) CONSTRAINT NOT NULL, -- chyba 2 - chybějící název omezení (constraint)
CONSTRAINT emailclinets_email2 NOT NULL, -- chyba 3 - NOT NULL nelze vytvářet na úrovni tabulky
CONSTRAINT cleints_clinet_num_pk2 PRIMARY KEY (client_number)
);
--SHRNUTÍ OMEZENÍ
-- not null - lze ho vytvářet pouze na úrovni sloupce, slouží pro kontrolu prázdných hodnot,
-- pokud má sloupec omezení not null, je nutné vždy vložit nějakou hodnotu
-- unique - lze ho vytvářet jak na úrovni sloupce, tak na úrovni tabulky a slouží pro odstranění
-- duplicitních hodnot ve sloupcích již na vstupu při tvorbě záznamů
-- primary key - lze ho vytvářet jak na úrovni sloupce, tak na úrovni tabulky a slouží pro jednoznačenou identifikaci záznamů v tabulce, tabulka má právě jeden primární klíč, primární klíč může být složen z jednoho nebo více sloupců

--FOREIGN KEY - referenční integrita
-- Máme-li tabulku DEPARTMENTS, která reprezentuje rodiče a tabulku EMPLOYEE,
-- která reprezentuje potomka. V tabulce DEPARTMENTS (rodič) máme určen primární klíč ve sloupci
-- DEPARTMENT_ID a v tabulce EMPLOYEE (potomek) máme určen cizí klíč ve sloupci DEPARTMENT_ID.
-- Pokud bychom chtěli odstranit záznamy z tabulky DEPARTMENTS, na které je navázaný libovolný
-- záznam z tabulky EMPLOYEE nebylo by takové odstranění databázovým systémem povoleno,
-- protože se jedná právě o porušení referenční integrity dat.
-- Pokud bychom chtěli odstranit záznamy z tabulky EMPLOYEE, tak je výmaz povolen.
-- PK nesmí obsahovat hodnotu NULL, FK může obsahovat hodnotu NULL. Pro hodnotu FK platí,
-- že musí existovat daná hodnota ve sloupci PK druhé tabulky.
-- Nejprve musí být vytvořen PK, který bude referencován FK.
-- FK lze vytvářet, jak na úrovni sloupce, tak na úrovni tabulky.

--tvorba FK na úrovni sloupce
CREATE TABLE copy_employees1
(
employee_id NUMBER(6,0) CONSTRAINT copy_emp_pk1 PRIMARY KEY,
