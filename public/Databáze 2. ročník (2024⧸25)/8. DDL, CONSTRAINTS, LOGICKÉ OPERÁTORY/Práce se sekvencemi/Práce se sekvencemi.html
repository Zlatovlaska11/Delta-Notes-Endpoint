--vyzkoušíme si vytvoření vlastní sekvence
CREATE SEQUENCE mojesekvence
INCREMENT BY 1
START WITH 1;
--u sekvencí je increment by a start with defaultně nastaveno
CREATE SEQUENCE mojesekvence2;
--vytvorime si tabulku uzivatele1
CREATE TABLE uzivatele1 (
    ID NUMBER CONSTRAINT uzivatele_pk PRIMARY KEY,
    jmeno VARCHAR2(50) CONSTRAINT uziv_jmeno_nn NOT NULL,
    prijmeni VARCHAR2(50) CONSTRAINT uziv_prijmeni_nn NOT NULL, 
    email VARCHAR2(100) CONSTRAINT uziv_email_un UNIQUE NOT NULL, 
    datumRegistrace DATE DEFAULT SYSDATE,
    preferovanyJazyk VARCHAR2(20) DEFAULT 'Czech'
);
--k sekvenci mojesekvence vytvoříme trigger a ten použijeme na tabulku uzivatele1
CREATE OR REPLACE TRIGGER trg_uzivatele1_id
BEFORE INSERT ON uzivatele1
FOR EACH ROW
BEGIN
  IF :NEW.ID IS NULL THEN
    :NEW.ID := mojesekvence.NEXTVAL;
  END IF;
END;
