1. Vytvořte proceduru

create or replace PROCEDURE delete_script(p_schema VARCHAR2) AS

   CURSOR c_script IS

    SELECT 

      'drop table '||owner||'.'||table_name||' cascade constraints;'

    FROM

      all_tables

    WHERE

      UPPER(owner) = UPPER(p_schema) 

    ORDER BY

      table_name;

  i_script_row VARCHAR2(500);

BEGIN

  OPEN c_script;

  LOOP

    FETCH c_script INTO i_script_row;

    EXIT WHEN c_script%NOTFOUND; 

    dbms_output.put_line(i_script_row);

  END LOOP;

  CLOSE c_script;  

END delete_script;

/


2. Spusťte proceduru a jako parametr dejte do uvozovek Vaši konkrétní databázi

begin

delete_script ('');

end;
