SELECT 4+5, (6/2)*3, 6*5 FROM DUAL;

--1. varianta u ALIAS (jednoslovný název, může být i diakritika)

SELECT 4+5 Prvni, (6/2)*3 Druhy, 6*5 Treti FROM DUAL;

SELECT 4+5 První, (6/2)*3 Druhý, 6*5 Třetí FROM DUAL;

--2. varianta u ALIAS (víceslovný název)

SELECT 4+5 "První sloupec", (6/2)*3 "Druhý sloupec",

6*5 "Třetí sloupec" FROM DUAL;

--3. varianta u ALIAS (použiji AS)

SELECT 4+5 AS První, (6/2)*3 AS Druhý, 6*5 AS Třetí FROM DUAL;

SELECT 4+5 AS "První sloupec", (6/2)*3 AS "Druhý sloupec",

6*5 AS "Třetí sloupec" FROM DUAL;
