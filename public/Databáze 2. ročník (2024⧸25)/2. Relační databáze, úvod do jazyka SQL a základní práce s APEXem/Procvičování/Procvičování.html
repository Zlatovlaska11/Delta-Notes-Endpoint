--ora-00900 - invalid SQL statement (nesprávný SQL příkaz - syntaxe)

--ora-00942 - table or view does not exist (chybný název tabulky)

SELECT * FROM employees;

--selekce s podmínkou

SELECT * FROM employees WHERE last_name LIKE 'King';

--ukázka s jinou tabulkou

SELECT * FROM countries;

--výčet sloupců

SELECT country_id, country_name, region_id FROM countries;

--selekce s podmínkou

SELECT * FROM countries WHERE country_name LIKE 'Germany';

-- projekce vybraných sloupců z tabulky countries

SELECT country_id, country_name FROM countries; 

-- selekce (vybírám pouze řádky, které splňují podmínku za klíčovým slovem WHERE)

SELECT country_id, country_name FROM countries WHERE country_name LIKE 'United Kingdom';

--použití aritmetických operací

SELECT last_name, salary, salary+300 FROM employees;

SELECT last_name, salary, 12*salary+100 FROM employees;

SELECT last_name, salary, 12*(salary+100)FROM employees;

--použití aritmetických operací s ALIASY

--víceslovný alias musí mít uvozovky

SELECT last_name, salary, salary+300 "Zvýšený plat o 300" FROM employees;

--u aliasu můžete použít AS (v některých případech je nutné jej použít - více později)

SELECT last_name, salary, salary+300 AS "Zvýšený plat o 300" FROM employees;

--takto použitý alias je nefunkční, jelikož se jedná o víceslovný a není ve dvojitých uvozovkách

SELECT last_name, salary, salary+300 Zvýšený plat o 300 FROM employees;

--ukázka aliasy

SELECT last_name, salary, 12*salary+100 FROM employees;

SELECT last_name, salary, 12*salary+100 "Roční plat zvýšení o 100" FROM employees;

SELECT last_name, salary, 12*salary+100 AS "Roční plat zvýšení o 100" FROM employees;

SELECT last_name, salary, 12*(salary+100) Plat FROM employees;

--null hodnoty

SELECT last_name, job_id, salary, commission_pct, salary*commission_pct FROM employees;

--aliasy

SELECT last_name AS name, commission_pct AS comm FROM employees;

SELECT last_name "Name", commission_pct "comm" FROM employees;

SELECT last_name Name, commission_pct comm FROM employees;

SELECT last_name "Moje příjmení" FROM employees;

SELECT last_name Moje příjmení FROM employees; --chyba ORA-00923, alias není správně zapsán, není znám konec a začátek aliasu

SELECT last_name AS Moje příjmení FROM employees; -- chyba ORA-00923, alias není správně zapsán, není znám konec a začátek aliasu

SELECT last_name AS "Moje příjmení" FROM employees;
