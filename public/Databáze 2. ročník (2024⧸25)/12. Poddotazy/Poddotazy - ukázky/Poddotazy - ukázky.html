Poddotazy - ukázky

-- Obecně k jednořádkovým dotazům

--- pracujeme s vnějším a vnitřním dotazem

--- můžeme pracovat s daty z rozdílných tabulek

--- při použití ORDER BY platí, že klauzule může být použita pro příkaz SELECT, musí být poslední klauzuli v hlavním SELECTu příkazu


-- Jednořádkové poddotazy

--- používají jednořádkové porovnávací operátory (.,=,...)

--- vždy jsou uzavřeny v závorkách, vždy jsou umístěny na pravé straně porovnávací podmínky 


--- Jména zaměstnanců, kteří do firmy přišli až po Peterovi Vargasovi.

SELECT first_name, last_name, hire_date

FROM employees

WHERE hire_date > (SELECT hire_date FROM employees WHERE last_name LIKE 'Vargas');


--- Příklad


---- Kdo pracuje na marketingovém oddělení?

SELECT last_name, job_id, department_id

FROM employees

WHERE department_id = (SELECT department_id FROM departments WHERE department_name LIKE 'Marketing')

ORDER BY job_id;


--- Group funkce v poddotazech

---- Kteří zaměstnanci si vydělají méně než je průměrný plat?

SELECT last_name, salary

FROM employees

WHERE salary < (SELECT AVG(salary) FROM employees);


--- Poddotazy v klauzule HAVING

---- Která oddělení mají nižší plat než nejnižší plat v oddělení 50?

SELECT department_id, MIN(salary)

FROM employees

GROUP BY department_id

HAVING MIN(salary) > (SELECT MIN(salary) FROM employees WHERE department_id=50);



-- Víceřádkové poddotazy

--- nemůžeme použít jednořádkový porovnávací operátor (=,<,>,...)

--- jako operátory používáme IN, ANY, ALL, NOT

--- Pokud budeme chtít vědět, kdo má plat shodný s platem zaměstnance z oddělení 20,

potom hodí dotaz chybu. Je to z důvodů, že výsledná množina z poddotazu obsahuje

více než jeden záznam.


SELECT first_name, last_name

FROM employees

WHERE salary = (SELECT salary FROM employees WHERE department_id=20);


SELECT first_name, last_name, salary

FROM employees

WHERE salary IN (SELECT salary FROM employees WHERE department_id=20);


SELECT salary FROM employees WHERE department_id=20; --6000, 13000


--- Poznámka: Problémem je zde znaménko = v klauzuli where.


--- Jedna hodnota nemůže být rovna více než jedné hodnotě na právě straně.

SELECT EXTRACT(YEAR FROM hire_date)

FROM employees

WHERE department_id=90;

--vrací 1987, 1989, 1993

SELECT last_name, EXTRACT(YEAR FROM hire_date), hire_date

FROM employees;

--vrací 1987,1989,1990,1991,1993,1994,1995,1996,1997,1998,1999,2000,


SELECT last_name, hire_date

FROM employees

WHERE EXTRACT(YEAR FROM hire_date) IN

(SELECT EXTRACT(YEAR FROM hire_date)

FROM employees

WHERE department_id=90);



--- Vráťte zaměstnance, jejichž rok přijetí je menší nebo roven alespoň jednomu roku přijetí zaměstnanců z oddělení 90.

SELECT last_name, hire_date

FROM employees

WHERE EXTRACT(YEAR FROM hire_date) < ANY

(SELECT EXTRACT(YEAR FROM hire_date)

FROM employees

WHERE department_id=90);

--vnitřní vrací 1987, 1989, 1993



SELECT last_name, hire_date

FROM employees

WHERE EXTRACT(YEAR FROM hire_date) < ALL

(SELECT EXTRACT(YEAR FROM hire_date)

FROM employees

WHERE department_id=90);


--- NULL hodnoty

IN a ANY - vrací ne-null hodnoty

ALL - vnější dotaz nevrátí žádné řádky, protože bude porovnávat hodnoty včetně null hodnot. Když porovnáváme cokoliv s null, výsledek je null.


--Vícesloupcové poddotazy

--- Zobrazte seznam zaměstnanců, jejichž manažerem a odděleními jsou ty samé jako manažeři a oddělení zaměstnanců 149 nebo 174.

SELECT employee_id, manager_id, department_id

FROM employees

WHERE (manager_id, department_id) IN

(SELECT manager_id, department_id

FROM employees

WHERE employee_id IN (149,174))

AND employee_id NOT IN (149,174);

--department_id = 80, manager_id = 100,149

--pouze jeden zaměstnanec, a to 176



-- EXIST, NOT EXISTS


SELECT last_name AS "No a Manager"

FROM employees emp

WHERE NOT EXISTS

(SELECT * FROM employees mgr

WHERE mgr.manager_id=emp.employee_id);




-- Korelované podotazy

SELECT o.first_name, o.last_name, o.salary o

FROM employees o

WHERE o.salary >

(SELECT AVG(i.salary) FROM employees i

WHERE i.department_id=o.department_id);



--WITH


WITH managers AS

(SELECT DISTINCT manager_id

FROM employees WHERE

manager_id IS NOT NULL)

SELECT last_name AS "Not a manager"

FROM employees

WHERE employee_id NOT IN

(SELECT * FROM managers);
