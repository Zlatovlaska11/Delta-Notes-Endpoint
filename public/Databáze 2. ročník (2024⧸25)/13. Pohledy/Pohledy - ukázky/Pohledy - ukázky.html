--ukázka tvorby pohledu 4 - komplexní pohled

CREATE OR REPLACE VIEW view_euro_countries

("ID", "Country", "Capitol City", "Region")

AS SELECT c.country_id, c.country_name, c.capitol, r.region_name

FROM wf_countries c JOIN wf_world_regions r

USING (region_id)

WHERE location LIKE '%Europe';


--výpis dat z pohledu 4

SELECT * from view_euro_countries;


--ukázka tvorby pohledu  - komplexní pohledy

CREATE OR REPLACE VIEW view_high_pop

("Region ID", "Highest population")

AS SELECT region_id, MAX(population)

FROM wf_countries

GROUP BY region_id;


SELECT * FROM view_high_pop;

-- DML operace nad pohledy, mohou být použity nad jednoduchými pohledy

CREATE TABLE copy_employees AS (SELECT * FROM employees);


CREATE OR REPLACE VIEW view_dept50

AS SELECT department_id, employee_id, first_name, last_name, salary

FROM copy_employees

WHERE department_id=50;


SELECT * FROM view_dept50;

-- update skrz pohled view_dept50


UPDATE view_dept50

SET department_id=90

WHERE employee_id=124;


/*možnost with check option kontroluje, že DML operace prováděné na pohledu zůstávají pouze v doméně pohledu.

Jakýkoliv jiný pokus o změnu skončí chybou.*/


CREATE OR REPLACE VIEW view_dept50

AS SELECT department_id, employee_id, first_name, last_name, salary

FROM employees

WHERE department_id=50

WITH CHECK OPTION CONSTRAINT view_dept50_check;


--pokus o UPDATE upravené pohledu view_dept50

UPDATE view_dept50

SET department_id=90

WHERE employee_id=124;


--možnost read only zajistí, že žádné DML operace nemohou být nad pohledem prováděny

CREATE OR REPLACE VIEW view_dept50

AS SELECT department_id, employee_id, first_name, last_name, salary

FROM employees

WHERE department_id=50

WITH READ ONLY;


--odstranění pohledu pomocí příkazu DROP VIEW

DROP VIEW view_dept50;


--inline pohledy, tyto pohledy jsou referovány jako poddodazy z klauzule from. Když vložíte poddotaz do from klauzule tak jenom, když má poddotaz název. Inline pohledy jsou používány pro zjednodušení komplexních dotazů odstraněním join operací a spojení několika dotazů do jednoho.

SELECT e.last_name, e.salary, e.department_id, d.maxsal

FROM employees e,

    (SELECT department_id, max(salary) maxsal

    FROM employees

    GROUP BY department_id) d

WHERE e.department_id=d.department_id

AND e.salary=d.maxsal;


--TOP-N-ANALYSIS

-- použití této analýzy je vhodné, když chcete například získat prvních 5 záznamů apod.

SELECT ROWNUM AS "Longest employed", last_name, hire_date

FROM employees

WHERE ROWNUM<=5

ORDER BY hire_date;
