--MERGE

CREATE TABLE copy_emp AS (SELECT * from copy_employees);


SELECT * FROM copy_emp;


MERGE INTO copy_emp c USING employees e

ON (c.employee_id=e.employee_id)

WHEN MATCHED THEN UPDATE SET (c.employee_id=e.employee_id, c.last_name=e.last_name,
c.email=e.email, c.hire_date=e.hire_date, c.job_id=e.job_id, c.department_id=e.department_id)

WHEN NOT MATCHED THEN INSERT VALUES (e.employee_id, e.last_name, e.email,
e.hire_date, e.job_id, e.department_id);
