Seskupovací funkce - ukázky


--seskupovací funkce nelze zapisovat do klauzule WHERE!

--chybová hláška: ORA-00934: group function is not allowed here

SELECT last_name, first_name

FROM employees

WHERE salary = MIN(salary);


--funkce MIN

SELECT MIN(life_expect_at_birth) AS "Lowest Life Exp" FROM wf_countries;--32.62

SELECT life_expect_at_birth AS "Lowest Life Exp" FROM wf_countries ORDER BY life_expect_at_birth;

SELECT MIN(country_name) FROM wf_countries;--Anguilla

SELECT country_name FROM wf_countries ORDER BY country_name;--. pozice Anguilla

SELECT MIN(hire_date) FROM employees;--17-Jun-1987

SELECT hire_date FROM employees ORDER BY hire_date;--17-Jun-1987


--funkce MAX

SELECT MAX(life_expect_at_birth) AS "Lowest Life Exp" FROM wf_countries;--83.51

SELECT MAX(country_name) FROM wf_countries;--Western Sahara

SELECT MAX(hire_date) FROM employees;--29-Jan-2000


--funkce SUM

SELECT SUM(area) FROM wf_countries WHERE region_id=29;--241424

SELECT SUM(salary) FROM employees WHERE department_id=90;--58000

-- ukázka chybové hlášky v případě použití sloupce s nevhodným datovým typem pro funkci SUM: 

-- ORA-00932: inconsistent datatypes: expected NUMBER got DATE

SELECT SUM(hire_date) FROM employees;


--funkce AVG

SELECT AVG(area) FROM wf_countries WHERE region_id=29;--9656.96

SELECT AVG(salary) FROM employees WHERE department_id=90;--19333.3333333333333333333333333333333333

SELECT ROUND(AVG(salary),2) FROM employees WHERE department_id=90;--19333.33


--funkce variance, stddev

SELECT ROUND(VARIANCE(life_expect_at_birth),4) FROM wf_countries;--143.2394

SELECT ROUND(STDDEV(life_expect_at_birth),4) FROM wf_countries;--11.9683


--Group funkce ingorují NULL hodnoty!

SELECT COUNT (*) FROM employees;--počet řádků v tabulce employees je 20

SELECT DISTINCT COUNT(job_id) FROM employees;--zjisti jedinečný počet job_id, ale je nutné zvážit smysl


--funkce NVL

--a) chyba 1


--chybová hláška: ORA-00937: not a single-group group function


SELECT department_id, AVG(salary) FROM employees ORDER BY department_id;--je nutné využít GROUP BY


--b) chyba 2


--ORA-00979: not a GROUP BY expression


SELECT department_id, salary FROM employees GROUP BY department_id ORDER BY department_id;--nelze


--správně


SELECT department_id, AVG(salary) FROM employees GROUP BY department_id ORDER BY department_id;


--c) chyba 3


--group funkce vyžaduje, aby každý sloupec použitý v select a který není součástí group funkce, byl uveden v klauzuli group by


--jinak vrací ORA-00979: not a GROUP BY expression


SELECT job_id, department_id, AVG(salary) FROM employees GROUP BY department_id ORDER BY department_id;


--správně, všimněte si, jak se změnil výpočet AVG

SELECT job_id, department_id, AVG(salary) FROM employees GROUP BY department_id, job_id ORDER BY department_id;


--další úkazka zápisu a práce s GROUP BY:

SELECT department_id, MAX(salary) FROM employees WHERE last_name !='King' GROUP BY department_id;


--pravidla GROUP BY

1. Když zahrnujete seskupovací funkci v selectu spolu s nějakými dalšími sloupci, každý jednotlivý sloupec se musí také objevit v group by klauzuli.

2. Nemůžete používat aliasy sloupců v klauzuli group by.

3. Klauzule where vyčleňuje řádky ještě před tím, než jsou rozděleny do skupin.

4. Funkce mohou být vnořeny do sebe, lze použít 2 funkce ihned po sobě. 


SELECT max(avg(salary)) FROM employees GROUP BY department_id; 



--having = filtrování záznamů ve skupině

SELECT department_id, MAX(salary) FROM employees GROUP BY department_id ORDER BY department_id;--8 řádků

SELECT department_id, MAX(salary) FROM employees GROUP BY department_id HAVING COUNT(*)>1 ORDER BY department_id;--6 řádků

--28 řádků, region_id=5,9,11,21,29,35,39,154 (8 regionů)

SELECT region_id, population from wf_countries WHERE population<30000 ORDER BY region_id;

--19 řádků dle region_id

SELECT region_id, ROUND(AVG(population)) FROM wf_countries GROUP BY region_id ORDER BY region_id;

--11 řádků dle regionů, které splňují podmínku v HAVING

SELECT region_id, ROUND(AVG(population)) FROM wf_countries GROUP BY region_id HAVING MIN(population)>30000 ORDER BY region_id;


--ROLLUP

SELECT department_id, job_id, SUM(salary) FROM employees WHERE department_id<50 GROUP BY ROLLUP (department_id, job_id);

-- DEPARTMENT_ID JOB_ID SUM(SALARY)

-- 10 AD_ASST 4400 - první řádek aplikuje agregační funkci SUM dle kategorie, a to department_id a pak dle job_id

-- 10 - 4400 - druhý řádek je mezisoučet za department_id = 10

-- 20 MK_MAN 13000 - třetí řádek aplikuje agregační funkci SUM dle kategorie, a to department_id a pak dle job_id

-- 20 MK_REP 6000 - čtvrtý řádek aplikuje agregační funkci SUM dle kategorie, a to department_id a pak dle job_id

-- 20 - 19000 - pátý řádek je mezisoučet za department_id = 20

-- - - 23400 - šestý řádek je součet mezisoučtů za všechna department_id

--bez použití ROLLUP, je nutné použít vícero dotazů k získání souhrnu produkovaných operací rollup

SELECT department_id, job_id, SUM(salary) FROM employees WHERE departmetn_id<50 GROUP BY (department_id, job_id);


--CUBE

SELECT department_id, job_id, SUM(salary) FROM employees WHERE department_id<50 GROUP BY CUBE (department_id, job_id);

-- DEPARTMENT_ID JOB_ID SUM(SALARY)

-- - - 23400 -- první řádek vrací součet mezisoučtů za všechna oddělení

-- - MK_MAN 13000 -- druhý řádek vrací mezisoučet za pozici MK_MAN

-- - MK_REP 6000 -- třetí řádek vrací mezisoučet za pozici MK_REP

-- - AD_ASST 4400 -- čtvrtý řádek vrací mezisoučet za pozici AD_ASST

-- 10 - 4400 -- pátý řádek vrací součet mezisoučtů za pozice v oddělení 10

-- 10 AD_ASST 4400 -- šestý řádek vrací mezisoučet za pozici AD_ASST v oddělení 10

-- 20 - 19000 -- sedmý řádek vrací mezisoučty za pozice v oddělení 20

-- 20 MK_MAN 13000 -- osmý řádek vrací mezisoučet za pozici MK_MAN v oddělení 20

-- 20 MK_REP 6000 -- devátý řádek vrací mezisoučet za pozici MK_REP v oddělení 20

--GROUPING SETS

SELECT department_id, job_id, manager_id, SUM(salary)

FROM employees

WHERE department_id < 50

GROUP BY GROUPING SETS ((job_id, manager_id),(department_id, job_id),(department_id, manager_id));
