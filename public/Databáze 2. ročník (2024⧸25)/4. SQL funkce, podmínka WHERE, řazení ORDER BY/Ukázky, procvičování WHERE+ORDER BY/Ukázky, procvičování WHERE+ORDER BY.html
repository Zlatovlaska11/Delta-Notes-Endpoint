/*zřetězení s mezerou*/

/*zřetězení s literálními hodnotami a s aliasem s použitím AS*/

SELECT last_name ||' has a '|| 1 ||' year salary of '|| salary*12 ||' dollars.' AS Pay FROM employees;

/*zřetězení s literálními hodnotami bez aliasu*/

SELECT last_name ||' has a '|| 1 ||' year salary of '|| salary*12 ||' dollars.' FROM employees;

/*zřetězení s literálními hodnotami a s aliasem*/

SELECT last_name ||' has a '|| 1 ||' year salary of '|| salary*12 ||' dollars.' Pay FROM employees;

/*zobrazení všech department_id*/

select department_id from employees;

/*zobrazení department_id bez opakování, tedy zobrazení každého department_id jen jednou*/

select distinct department_id from employees;

/*výběr jen zaměstnance, který má employee_id=102*/

select employee_id, first_name, last_name from employees where employee_id=102;

/*výběr zaměstnanců, kteří nastoupili před 1.1.2000*/

select * from employees WHERE hire_date < '01-Jan-2000';

/*výběr zaměstnanců, jejichž plat je roven nebo vyšší než 6000*/

select * from employees WHERE salary >= 6000;

/*výběr zaměstnanců jejich pracovní pozice je IT_PROG*/

select * from employees WHERE job_id ='IT_PROG';

/*chyba při neuvedení stringu do uvozovek - ORA-00904: "IT_PROG": invalid identifier*/

select * from employees WHERE job_id =IT_PROG;

/*výběr zaměstananců, jejichž plat je v rozemí 9-11 tis.(>= x a zároveň <=)*/

select * from employees WHERE salary BETWEEN 9000 AND 11000;

/*výběr zaměstananců, jejichž plat je v rozemí 9-11 tis.(>= x a zároveň <=); stejný požadavek jako výše, ale jinak zapsaný, datový set stejný*/


select * from employees WHERE salary>=9000 AND salary <=11000;


--operátor IN (logické OR)


/*výběr lokací, kde je country_id 44 nebo 55*/


select city, state_province, country_id FROM locations WHERE country_id IN('CA','UK');


/*výběr lokací, kde je country_id 44 nebo 55*; stejný požadavek jako výše, ale jinak zapsaný, datový set stejný*/


select city, state_province, country_id FROM locations WHERE country_id='CA' OR country_id='UK';


--LIKE, _ nahrazuje jeden znak na dané pozici, % nahrazuje více znaků


/*výběr zaměstnanců, jejichž příjmení má na druhé pozici o*/


select * from employees where last_name LIKE '_o%';


--IS NULL, IS NOT NULL


/*výběr zaměstanců, kteří nemají nadřízeného, manager_id obsahuje NULL hodnotu*/


SELECT last_name, manager_id FROM employees WHERE manager_id IS NULL;


/*výběr zaměstnanců, kteří jsou odměňování provizemi, jejichž provize neobsahuje NULL hodnotu*/


SELECT last_name, commission_pct from employees WHERE commission_pct IS NOT NULL;


/*výběr zaměstnanců, kteří spadají do oddělení s označením větším než 50 a jejichž plat je větší než 12000*/


select last_name, department_id, salary FROM employees WHERE department_id>50 AND salary>12000;


/*výběr zaměstnanců, kteří spadají do oddělení s označením větším než 20 nebo jejichž plat je větší než 12000*/


select last_name, department_id, salary FROM employees WHERE department_id>20 OR salary>12000;


/*výběr oddělení, jejichž lokace je 2500 nebo jejichž nadřízený je veden pod id=124*/


select department_name, manager_id, location_id FROM departments WHERE location_id=2500 OR manager_id=124;


/*výběr oddělení, jejiž lokace je jiná než 1700 nebo 1800*/


select department_name, location_id FROM departments WHERE location_id NOT IN (1700,1800);

--order by

/*vypiště zaměstnance vzestupně dle jejich data nástupu - 2 varianty zápisu, stejný výstupní datový set*/

select last_name, hire_date FROM employees ORDER BY hire_date; --výchozí vzestupně

select last_name, hire_date FROM employees ORDER BY hire_date ASC; --vzestupně

/*vypiště zaměstnance sestupně dle jejich data nástupu, tedy od těch, kteří pracují v organizaci nejkratší dobu*/

select last_name, hire_date FROM employees ORDER BY hire_date DESC; --sestupně

/*výběr zaměstnanců vzestupně dle jich data nástupu s využitím aliasu v order by*/

select last_name, hire_date AS "Date Started" FROM employees ORDER BY "Date Started";

/*výběr zaměstnanců seřazených vzestupně podle jejich příjmení, a to pouze zaměstnanců, kteří mají identifikátor větší než 105*/

select employee_id, first_name FROM employees WHERE employee_id < 105 ORDER BY last_name;

--řazení podle více sloupců

/*vypiště zaměstnance, kteří pracují v oddělení s označením 50 a níže, zaměstnance vypište dle označení oddělení sestupně a v rámci oddělení vzestupně podle jejich příjmení*/

select department_id, last_name FROM employees WHERE department_id<=50 ORDER BY department_id DESC, last_name;
