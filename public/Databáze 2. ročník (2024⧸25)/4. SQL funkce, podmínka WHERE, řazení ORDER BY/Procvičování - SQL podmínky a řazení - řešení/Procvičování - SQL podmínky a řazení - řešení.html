mezi 5 a 10, a to jen s pomocí pojmenovaných operátorů porovnání*/

/*7. Popřemýšlejte o rozdílu níže uvedených příkazů a přepište oba dotazy s využitím jen logických operátorů.*/
SELECT code, description FROM d_themes WHERE code>200 AND description IN('Tropical', 'Football', 'Carnival');
--vrací tento výstupní datový set
/*
CODE	DESCRIPTION
220	Carnival
340	Football
*/
--správný přepis pro AND
SELECT code, description FROM d_themes WHERE (code>200) AND (description LIKE 'Tropical' OR description LIKE 'Football' OR description LIKE 'Carnival');
--POZOR, tento zápis vrací jiný výstupní datový set
SELECT code, description FROM d_themes WHERE code>200 OR description IN('Tropical', 'Football', 'Carnival');
/*
CODE	DESCRIPTION
200	Tropical
220	Carnival
240	Sixties
454	Eighties
340	Football
502	Fairy Tale
*/
--správný přepis
SELECT code, description FROM d_themes WHERE (code>200)
OR (description LIKE 'Tropical' OR description LIKE 'Football' OR description LIKE 'Carnival');
--Rozdíl v dotazech. V případě AND musí platit obě podmínky, v případě OR stačí, že platí alespoň jedna podmínky.
/*8. Zobrazte last_name z tabulky f_staffs, kde příjmení obsahuje znak 'e' a zároveň obsahuje znak 'i'.*/

/*9. Zobrazte last_name zaměstnance z tabulky f_staffs, který má plat větší než 6.50 a jejich pozice (staff_type)
není Order Taker*/

SELECT last_name FROM f_staffs WHERE salary>6.50 AND staff_type NOT LIKE 'Order Taker';
--v tomto případě si můžeme dovolit použití i nerovná se, ale spíše nedoporučuji
SELECT last_name FROM f_staffs WHERE salary>6.50 AND staff_type!='Order Taker';
SELECT last_name FROM f_staffs WHERE salary>6.50 AND staff_type <> 'Order Taker';

/*17. Napište dotaz, který zobrazí zaměstnance, kteří mají employee_id větší než 125. Zobrazte u takových zaměstnanců department_id, last_name, manager_id a seřaďte záznamy sestupně nejdříve podle department_id, pak last_name a dále podle manager_id.*/
