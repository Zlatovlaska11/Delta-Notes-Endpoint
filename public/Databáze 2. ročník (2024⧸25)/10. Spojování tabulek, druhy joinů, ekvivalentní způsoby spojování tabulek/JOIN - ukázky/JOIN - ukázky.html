--1) cross join, což je kartézský součin

SELECT last_name, department_name

FROM employees CROSS JOIN departments 

ORDER BY last_name;

-- počet řádků ve výstupním data setu z cross joinu

select count(*) from departments;--8

select count(*) from employees;--20

select 8 * 20 from dual;--160 řádků vrátí cross join

-- 2) cross join tedy kartézský součin bez použití CROSS JOIN

SELECT last_name, department_name

FROM employees, departments

ORDER BY last_name;

-- 3) natural join tedy přirozené spojení, spojuje se na základě shody názvu sloupce z obou tabulek + datový typ u tohoto sloupce musí být v obou tabulkách shodný

SELECT department_id, department_name, location_id, city

FROM departments NATURAL JOIN locations;


--4) natural join s podmínkou (WHERE)

SELECT department_id, department_name, location_id, city

FROM departments NATURAL JOIN locations

WHERE department_id IN (20,50);

-- natural join s podmínkou stejný jako u 4) přičemž je zde rozepsaná podmínka s logickým operátorem místo IN

SELECT department_id, department_name, location_id, city

FROM departments NATURAL JOIN locations

WHERE department_id=20 OR department_id=50;

--5) vnitřní spojení - JOIN USING (Oracle), u použití USING musí být shodný název sloupce v obou tabulkách

SELECT l.city, d.department_name

FROM locations l JOIN departments d

USING (location_id)

WHERE location_id = 1400;

--6) vnitřní spojení - JOIN ON, za ON se napíše vyhledávací podmínky, tedy na jakém základě se mají tabulky spojit, kde má docházet například k rovnosti sloupců

SELECT l.city, d.department_name

FROM locations l JOIN departments d

ON l.location_id=d.location_id

WHERE l.location_id = 1400;

--7) vnitřní spojení - bez použití JOIN ON a bez použití JOIN USING, jedná se o equijoin

SELECT l.city, d.department_name

FROM locations l, departments d

WHERE l.location_id=d.location_id

AND l.location_id = 1400;

/*8) levé vnější spojení=LEFT OUTER JOIN či LEFT JOIN, při této vyhledávací podmínky (rovnost) dojde nejdříve k výpisu všech záznamů, kde existuje shoda

hodnot ve sloupci department_id v obou tabulkách a dále jsou vypsané všechny záznamy z tabulky employees a záznamy z tabulky employees, které nedokážeme spojit na základě vyhledávací podmínky (ON) jsou také vypsané a do sloupců z tabulky departments jsou doplněny null (APEX zobrazuje jako pomlčky), department_id může tak mít hodnotu null, tabulka employees má své PK a FK na tabulku departments(department_id) a tabulka departments má své PK (department_id).

*/

--vypíše všechny zaměstnance bez ohledu, zda patří do nějakého oddělení

SELECT e.first_name, e.last_name, d.department_id

FROM employees e

LEFT OUTER JOIN departments d

ON(e.department_id = d.department_id);

--9) LEFT OUTER JOIN s použitím (+) ORACLE (jen pro info, neučte se)

SELECT e.first_name, e.last_name, d.department_id

FROM employees e, departments d

WHERE e.department_id = d.department_id(+);

--10) RIGHT OUTER JOIN či RIGHT JOIN - vypíše všechny oddělení a to i takové, kde nepracuje žádný zaměstnanec, takže first_name a last_name může být null.

--vypíše všechna oddělení bez ohledu na to, zda v daném oddělení pracuje nějaký zaměstnanec

SELECT e.first_name, e.last_name, d.department_id

FROM employees e RIGHT OUTER JOIN departments d

ON(e.department_id = d.department_id);

--11) RIGHT OUTER JOIN s použitím (+) ORACLE (jen pro info, neučte se)

SELECT e.first_name, e.last_name, d.department_id

FROM employees e, departments d

WHERE( e.department_id(+) = d.department_id);

/*12) FULL OUTER JOIN, nejdříve vypíše záznamy, kde existuje shoda a dále doplní všechny záznamy z levé tabulky s možnými null a také záznamy z pravé tabulky s možnými null.*/

SELECT e.first_name, e.last_name, d.department_id

FROM employees e FULL OUTER JOIN departments d

ON (e.department_id = d.department_id);

--13) VNITŘNÍ SPOJENÍ - EQUIJOIN s restrikcí (IN)

SELECT s.title, s.artist, t.description

FROM d_songs s, d_types t

WHERE s.type_code = t.code AND s.ID IN(47,48);

--stejný výstupní datový set jako u 13) s tím, že zápis restrikce IN přímo za vyhledávací podmínkou bez WHERE

SELECT s.title, s.artist, t.description

FROM d_songs s JOIN d_types t

ON s.type_code = t.code AND s.ID IN(47,48);

--14) Kartézský součin s restrikcí (WHERE) 

SELECT e.last_name, e.salary, j.grade_level

FROM employees e, job_grades j

WHERE e.salary BETWEEN j.lowest_sal AND j.highest_sal;

--to samé jako 14) s použitím CROSS JOIN

SELECT e.last_name, e.salary, j.grade_level

FROM employees e CROSS JOIN job_grades j

WHERE e.salary BETWEEN j.lowest_sal AND j.highest_sal;

--15) (+) ORACLE - LEFT OUTER JOIN (jen pro info, neučte se)

SELECT e.last_name, e.department_id, d.department_name

FROM employees e, departments d

WHERE e.department_id = d.department_id (+); 

-- to samé jako 15)

SELECT e.last_name, e.department_id, d.department_name

FROM employees e LEFT OUTER JOIN departments d

ON e.department_id = d.department_id; 

-- 16) Vnitřní spojení JOIN ON, klasická shoda na základě rovnosti hodnot v obou tabulkách skrz vybraný sloupec (department_id)

SELECT e.last_name, e.department_id, d.department_name

FROM employees e JOIN departments d

ON e.department_id = d.department_id; 

-- 17) Přirozené spojení (automatické spojení skrz sloupec department_id)

SELECT last_name, department_id, department_name

FROM employees NATURAL JOIN departments;

--18) Vybere first_name, last_name, job_id, job_title s použitím NATURAL JOIN a podmínky, že se vypíší pouze záznamy, kde department_id>80

SELECT first_name, last_name, job_id, job_title 

FROM employees NATURAL JOIN jobs

WHERE department_id>80;

--19) Vybere first_name, last_name, department_id, department_name z tabulky employees a departments s použitím klauzule USING, a to pouze takové záznamy, kde příjmení zaměstnance rovná se Higgins.

SELECT first_name, last_name, department_id, department_name 

FROM employees JOIN departments USING(department_id)

WHERE last_name = 'Higgins';

/*jiný způsob zápisu, pozor na nejednoznačně definovaný sloupec (department_id), je nutné u tohoto sloupce, který je v obou tabulkách, uvést z jaké tabulky se mají hodnoty tohoto sloupce vypsat, platí při použití JOIN ON, v případě JOIN USING toto nemusíme řešit*/

--chybně

SELECT first_name, last_name, department_id, department_name

FROM employees JOIN departments

ON employees.department_id=departments.department_id

WHERE last_name = 'Higgins';

--správně

SELECT first_name, last_name, employees.department_id, department_name

FROM employees JOIN departments

ON employees.department_id=departments.department_id

WHERE last_name = 'Higgins';

/*jiný způsob zápisu s využitím equijoinu bez JOIN ON či JOIN USING*/

SELECT first_name, last_name, e.department_id, department_name 

FROM employees e, departments d

WHERE e.department_id=d.department_id AND last_name = 'Higgins';

--20) spojení více tabulek dohromady

SELECT last_name, department_name, city 

FROM employees JOIN departments USING(department_id)

JOIN locations USING (location_id);

--21) spojení s zřetězením hodnot ve výstupním datasetu

SELECT first_name || ' ' || last_name AS "Jméno a příjmení", department_name AS "Oddělení"

FROM employees JOIN departments USING (department_id);

--22) self join

SELECT e.last_name || ' pracuje pro ' ||m.last_name AS "Zaměstnanci a nadřízení"

FROM employees e JOIN employees m

ON (e.manager_id=m.employee_id)

ORDER BY e.last_name;

--23) hierarchický dotaz (rekurze)

SELECT level, employee_id, last_name, job_id, manager_id

FROM employees

START WITH employee_id=100

CONNECT BY PRIOR employee_id=manager_id

ORDER BY level;

--24) to samé jako 22 pomocí hierarchického dotazu

SELECT last_name || ' pracuje pro ' || PRIOR last_name AS "Zaměstnanci a nadřízení"

FROM employees

START WITH employee_id=100

CONNECT BY PRIOR employee_id=manager_id;

--25) hierarchický dotaz s použitím jednořádkových funkcí a klauzule PRIOR v SELECTu

SELECT LPAD(last_name, LENGTH(last_name)+(LEVEL*2)-2,'-') AS "Organizační struktura"

FROM employees

START WITH employee_id=100

CONNECT BY PRIOR employee_id=manager_id;
