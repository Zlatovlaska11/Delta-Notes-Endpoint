Formát výpisu desetinných čísel:


            double X;
            X = 1.1879;
            Console.WriteLine(X.ToString("F"));    // Displays 1.19
            Console.WriteLine(X.ToString("F0"));   // Displays 1

            X = -1.1987;
            Console.WriteLine(X.ToString("F1"));   // Displays -1.2

            Console.WriteLine(X.ToString("F3"));   // Displays -1,199
